Quick Start

Get Up & Running

SmartEcoHouse.ino — main sketch 
// Keyestudio Smart Eco-Friendly House — KS0576
// Requires: Adafruit_NeoPixel, DHT, LiquidCrystal_I2C

#include <Adafruit_NeoPixel.h>
#include <DHT.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// ── Pin Definitions ──────────────────────────────
#define RGB_PIN     11
#define PIR_PIN     3
#define LDR_PIN     A1
#define DHT_PIN     10
#define SOIL_PIN    A2
#define WATER_PIN   A3
#define RELAY_PIN   7
#define BUZZER_PIN  9
#define UV_PIN      A6
#define FAN_PIN     5

// ── Object Instantiation ─────────────────────────
Adafruit_NeoPixel rgb(1, RGB_PIN, NEO_GRB + NEO_KHZ800);
DHT dht(DHT_PIN, DHT11);
LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  Serial.begin(9600);
  rgb.begin(); dht.begin();
  lcd.init(); lcd.backlight();
  pinMode(PIR_PIN,    INPUT);
  pinMode(RELAY_PIN,  OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(FAN_PIN,    OUTPUT);
}

void loop() {
  float temp = dht.readTemperature();
  float humi = dht.readHumidity();
  int   soil = analogRead(SOIL_PIN);
  int   ldr  = analogRead(LDR_PIN);
  int   uv   = analogRead(UV_PIN);
  bool  pir  = digitalRead(PIR_PIN);

  // LCD: show temp, humidity, illuminance, UV
  lcd.setCursor(0, 0);
  lcd.print("T:"); lcd.print((int)temp);
  lcd.print(" H:"); lcd.print((int)humi);
  lcd.setCursor(0, 1);
  lcd.print("L:"); lcd.print(ldr);
  lcd.print(" UV:"); lcd.print(uv/100);

  // Auto-irrigation: pump on if soil is dry
  digitalWrite(RELAY_PIN, soil < 400 ? HIGH : LOW);

  // Smart fan: on if temp above 30°C
  analogWrite(FAN_PIN, temp > 30 ? 200 : 0);

  // Motion alarm: buzz if PIR triggered
  if (pir) { tone(BUZZER_PIN, 1000, 300); }

  delay(1000);
}
Installation
bash
# 1. Clone the repository
git clone https://github.com/keyestudio/KS0576-Keyestudio-Smart-Eco-Friendly-House-Kit

# 2. Open in Arduino IDE and install libraries:
#    Sketch → Include Library → Manage Libraries
#    - Adafruit NeoPixel
#    - DHT sensor library (Adafruit)
#    - LiquidCrystal I2C (Frank de Brabander)

# 3. Select board:  Tools → Board → Arduino UNO
# 4. Select port:   Tools → Port → (your COM port)
# 5. Upload:        Sketch → Upload  (Ctrl+U)
