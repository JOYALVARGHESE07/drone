// Mixer: Quadcopter X-configuration speedFL = constrain(throttle - pitchPID + rollPID, 1000, 2000); speedFR = constrain(throttle - pitchPID - rollPID, 1000, 2000); speedRL = constrain(throttle + pitchPID + rollPID, 1000, 2000); speedRR = constrain(throttle + pitchPID - rollPID, 1000, 2000); // Complementary filter for stable angle estimation angleRoll = 0.96 * (angleRoll + gyroRoll * dt) + 0.04 * accelRoll; anglePitch = 0.96 * (anglePitch + gyroPitch * dt) + 0.04 * accelPitch;
